Gửi ảnh đến Gemini Vision nhận diện (Tự động thử các phiên bản model tương thích) async function triggerAIScan() { if (isProcessing) return; const campaignId = document.getElementById("campaignSelect").value; if (!campaignId) { alert("Vui lòng chọn đợt phát quà trước!"); return; } isProcessing = true; showStatusCard("⏳ GEMINI AI ĐANG PHÂN TÍCH...", "Đang bóc tách mã QR và số in trên thẻ CCCD..."); // Chụp khung hình hiện tại từ camera canvas.width = video.videoWidth || 1280; canvas.height = video.videoHeight || 720; const ctx = canvas.getContext("2d"); ctx.drawImage(video, 0, 0, canvas.width, canvas.height); const base64Data = canvas.toDataURL("image/jpeg", 0.85).split(',')[1]; // Danh sách các model tương thích trên Google AI Studio const candidateModels = [ "gemini-2.5-flash", "gemini-2.0-flash", "gemini-1.5-flash-latest", "gemini-1.5-flash" ]; const requestBody = { contents: [{ parts: [ { text: "Đây là thẻ Căn cước công dân (CCCD) Việt Nam. Hãy đọc mã QR góc trên bên phải hoặc đọc dòng Số/No. in trên mặt thẻ. Bóc tách và trả về đúng định dạng JSON duy nhất: {\"cccd\": \"chuỗi 12 chữ số\", \"hoTen\": \"Họ và tên nếu có\"}. Nếu không thấy thẻ hoặc không đọc được, trả về {\"cccd\": null}. Tuyệt đối không thêm bất kỳ văn bản nào ngoài JSON." }, { inline_data: { mime_type: "image/jpeg", data: base64Data } } ] }], generationConfig: { response_mime_type: "application/json" } }; let successData = null; let lastErrorMessage = ""; // Vòng lặp tự động kết nối model sẵn có for (const modelName of candidateModels) { try { const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${GEMINI_API_KEY}`; const response = await fetch(apiUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(requestBody) }); const data = await response.json(); if (response.ok && data.candidates && data.candidates.length > 0 && data.candidates[0].content) { successData = data; break; // Tìm thấy model hoạt động tốt, thoát vòng lặp } else if (data.error) { lastErrorMessage = `Lỗi ${data.error.code} (${modelName}): ${data.error.message}`; if (data.error.code !== 404) { // Nếu không phải lỗi 404 (ví dụ hết hạn mức, sai khóa) thì dừng lại báo ngay break; } } } catch (e) { lastErrorMessage = e.message; } } if (!successData) { showErrorCard(lastErrorMessage || "Không kết nối được model AI của Google. Hãy thử lại!"); isProcessing = false; return; } try { const jsonText = successData.candidates[0].content.parts[0].text; const result = JSON.parse(jsonText); if (result && result.cccd && result.cccd.replace(/\D/g, '').length === 12) { const cleanCCCD = result.cccd.replace(/\D/g, ''); sendToGoogleSheets(cleanCCCD, result.hoTen || ""); } else { showErrorCard("Chưa thấy rõ số CCCD. Căn chỉnh thẻ vào giữa khung viền xanh và bấm quét lại!"); isProcessing = false; } } catch (parseErr) { showErrorCard("Lỗi định dạng kết quả AI: " + parseErr.message); isProcessing = false; } }
